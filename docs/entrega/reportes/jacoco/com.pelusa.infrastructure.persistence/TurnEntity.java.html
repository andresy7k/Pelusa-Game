package com.pelusa.infrastructure.persistence;

import jakarta.persistence.*;
import java.time.OffsetDateTime;
import java.util.UUID;

@Entity
@Table(name = "turns")
public class TurnEntity {
  @Id
  private UUID id;

  @Column(name = "game_id", nullable = false)
  private UUID gameId;

  @Column(name = "player_id", nullable = false)
  private UUID playerId;

  @Column(name = "started_at", nullable = false)
  private OffsetDateTime startedAt;

  /** Plazo del turno (ONLINE); null = sin límite (hotseat / turnos viejos). */
  @Column(name = "deadline_at")
  private OffsetDateTime deadlineAt;

  @Column(name = "ended_at")
  private OffsetDateTime endedAt;

  @Column(length = 16)
  private String outcome;

  @Column(columnDefinition = "TEXT")
  private String detail;

  @Column(nullable = false)
  private boolean acted;

  @Column(nullable = false)
  private boolean stashed;

  /** R-01/ADR-014: valor de la última carta sacada este turno (null = sin draw). */
  @Column(name = "last_drawn_value")
  private Integer lastDrawnValue;

  protected TurnEntity() {}

  public TurnEntity(UUID id, UUID gameId, UUID playerId) {
    this.id = id;
    this.gameId = gameId;
    this.playerId = playerId;
    this.startedAt = OffsetDateTime.now();
  }

  public UUID getPlayerId() { return playerId; }
  public OffsetDateTime getDeadlineAt() { return deadlineAt; }
  public void setDeadlineAt(OffsetDateTime deadlineAt) { this.deadlineAt = deadlineAt; }
  public String getDetail() { return detail; }
  public boolean isActed() { return acted; }
  public boolean isStashed() { return stashed; }
  public Integer getLastDrawnValue() { return lastDrawnValue; }
  public void setLastDrawnValue(Integer v) { this.lastDrawnValue = v; }
  public void setEndedAt(OffsetDateTime t) { this.endedAt = t; }
  public void setOutcome(String outcome) { this.outcome = outcome; }
  public void setDetail(String detail) { this.detail = detail; }
  public void setActed(boolean acted) { this.acted = acted; }
  public void setStashed(boolean stashed) { this.stashed = stashed; }
}
